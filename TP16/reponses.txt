Question 5:
Select id_adh, id_group from galette_groups_managers
union
(select g1.id_adh, g2.id_group from galette_groups_members g1
join galette_groups g2 on g1.id_group = g2.id_group
left join galette_groups_managers g3 on g2.id_group = g3.id_group
Where g3.id_group is null)
order by id_group
